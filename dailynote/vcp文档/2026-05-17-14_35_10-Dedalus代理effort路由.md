## 新增代码
```python
VALID_EFFORTS = {"none", "minimal", "low", "medium", "high", "xhigh"}
MODEL_SUFFIX_PATTERN = re.compile(r"^(.*?)\(([a-zA-Z]+)\)\s*$")

def parse_openai_model_with_effort(model_str):
    # 解析model(effort)语法，返回(clean_model, effort)
    if not isinstance(model_str, str) or not model_str:
        return (model_str, None)
    m = MODEL_SUFFIX_PATTERN.match(model_str)
    if not m:
        return (model_str, None)
    clean_model = m.group(1).strip()
    effort_raw = m.group(2).strip().lower()
    if effort_raw in VALID_EFFORTS:
        return (clean_model, effort_raw)
    log.warning("[ROUTE] Invalid effort '%s' in model '%s', falling back to default", effort_raw, model_str)
    return (clean_model, None)
```
